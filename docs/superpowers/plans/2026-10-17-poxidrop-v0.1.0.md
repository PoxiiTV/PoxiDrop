# PoxiDrop v0.1.0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir PoxiDrop, alternativa gratuita a Dropover: estantería flotante que aparece al agitar el ratón mientras arrastras, con acciones locales sobre los archivos.

**Architecture:** Paquete SwiftPM con dos módulos. `PoxiDropCore` contiene la lógica sin interfaz y con tests (detector de agitado, modelo, almacén, importador y acciones). `PoxiDrop` es la app de barra de menú: AppKit para ventanas flotantes, arrastre y menús, y SwiftUI para dibujar la estantería, el detalle, el buscador y los ajustes.

**Tech Stack:** Swift 6.4 (modo de lenguaje 5), SwiftPM, swift-testing, AppKit, SwiftUI, ImageIO, Vision, PDFKit, QuickLook, Carbon (atajos), ServiceManagement.

```swift
/// Trayectoria en zigzag entre 0 y `amplitude`, muestreada a 60 Hz.
func zigzag(amplitude: CGFloat, segments: Int, secondsPerSegment: Double) -> [(x: CGFloat, t: Double)] {
    let fps = 60.0
    let steps = max(1, Int(secondsPerSegment * fps))
    var points: [(x: CGFloat, t: Double)] = []
    var t = 0.0
    for segment in 0..<segments {
        let from: CGFloat = segment % 2 == 0 ? 0 : amplitude
        let to: CGFloat = segment % 2 == 0 ? amplitude : 0
        for step in 0..<steps {
            points.append((from + (to - from) * CGFloat(step) / CGFloat(steps), t))
            t += 1 / fps
        }
    }
    points.append((segments % 2 == 0 ? 0 : amplitude, t))
    return points
}

func detections(_ points: [(x: CGFloat, t: Double)], _ settings: ShakeDetector.Settings = .sensitivity(1)) -> Int {
    var detector = ShakeDetector(settings: settings)
    var count = 0
    for point in points where detector.add(x: point.x, time: point.t) {
        count += 1
    }
    return count
}
```

```swift
// FILE: Sources/PoxiDropCore/ShakeDetector.swift
import CoreGraphics

/// Detecta el gesto de agitar el ratón: varios cambios de dirección horizontal seguidos y rápidos.
public struct ShakeDetector: Sendable {
    public struct Settings: Equatable, Sendable {
        /// Cambios de dirección necesarios.
        public var reversals: Int
        /// Recorrido mínimo de cada tramo, en puntos.
        public var minSegment: CGFloat
        /// Segundos en los que tienen que caber todos los cambios.
        public var window: Double

        public init(reversals: Int, minSegment: CGFloat, window: Double) {
            self.reversals = reversals
            self.minSegment = minSegment
            self.window = window
        }

        /// 0 = baja, 1 = media, 2 = alta.
        public static func sensitivity(_ level: Int) -> Settings {
            switch level {
            case ..<1: return Settings(reversals: 4, minSegment: 60, window: 0.9)
            case 1: return Settings(reversals: 3, minSegment: 40, window: 0.8)
            default: return Settings(reversals: 2, minSegment: 30, window: 0.6)
            }
        }
    }

    public var settings: Settings
    private var lastX: CGFloat?
    private var direction: CGFloat = 0
    private var segmentStart: CGFloat = 0
    private var reversalTimes: [Double] = []

    public init(settings: Settings = .sensitivity(1)) {
        self.settings = settings
    }

    public mutating func reset() {
        lastX = nil
        direction = 0
        reversalTimes = []
    }

    /// Añade una posición horizontal del cursor. Devuelve `true` justo cuando se completa un agitado.
    public mutating func add(x: CGFloat, time: Double) -> Bool {
        guard let previous = lastX else {
            lastX = x
            segmentStart = x
            return false
        }
        let dx = x - previous
        guard abs(dx) >= 1 else { return false }
        lastX = x

        let newDirection: CGFloat = dx > 0 ? 1 : -1
        if direction == 0 {
            direction = newDirection
        } else if newDirection != direction {
            if abs(previous - segmentStart) >= settings.minSegment {
                reversalTimes.append(time)
            }
            direction = newDirection
            segmentStart = previous
        }

        reversalTimes.removeAll { time - $0 > settings.window }
        guard reversalTimes.count >= settings.reversals else { return false }

        reversalTimes = []
        direction = 0
        segmentStart = x
        return true
    }
}
```

```swift
// FILE: Tests/PoxiDropCoreTests/Helpers.swift
import Foundation
import CoreGraphics
import CoreText

/// Carpeta temporal para un test; se borra sola al terminar.
final class TempDir {
    let url: URL

    init() throws {
        let base = FileManager.default.temporaryDirectory.appendingPathComponent("poxidrop-tests-\(UUID().uuidString)", isDirectory: true)
        try FileManager.default.createDirectory(at: base, withIntermediateDirectories: true)
        url = base.resolvingSymlinksInPath()
    }

    deinit { try? FileManager.default.removeItem(at: url) }

    /// Crea un archivo de texto (y sus carpetas) dentro de la carpeta temporal.
    @discardableResult
    func file(_ path: String, _ contents: String = "hola") throws -> URL {
        let file = url.appendingPathComponent(path)
        try FileManager.default.createDirectory(at: file.deletingLastPathComponent(), withIntermediateDirectories: true)
        try contents.write(to: file, atomically: true, encoding: .utf8)
        return file
    }

    /// Crea una imagen de un solo color.
    @discardableResult
    func image(_ path: String, width: Int, height: Int, type: UTType = .png, properties: [CFString: Any]? = nil) throws -> URL {
        let context = Self.context(width: width, height: height)
        context.setFillColor(CGColor(red: 0.9, green: 0.3, blue: 0.4, alpha: 1))
        context.fill(CGRect(x: 0, y: 0, width: width, height: height))
        return try write(context.makeImage()!, path: path, type: type, properties: properties)
    }

    /// Imagen con texto negro grande sobre fondo blanco, para probar el OCR.
    func textImage(_ path: String, text: String) throws -> URL {
        let context = Self.context(width: 1000, height: 240)
        context.setFillColor(CGColor(gray: 1, alpha: 1))
        context.fill(CGRect(x: 0, y: 0, width: 1000, height: 240))
        let font = CTFontCreateWithName("Helvetica-Bold" as CFString, 96, nil)
        let attributed = NSAttributedString(string: text, attributes: [NSAttributedString.Key(kCTFontAttributeName as String): font])
        context.textPosition = CGPoint(x: 40, y: 80)
        CTLineDraw(CTLineCreateWithAttributedString(attributed), context)
        return try write(context.makeImage()!, path: path, type: .png, properties: nil)
    }

    private func write(_ image: CGImage, path: String, type: UTType, properties: [CFString: Any]?) throws -> URL {
        let file = url.appendingPathComponent(path)
        try FileManager.default.createDirectory(at: file.deletingLastPathComponent(), withIntermediateDirectories: true)
        guard let destination = CGImageDestinationCreateWithURL(file as CFURL, type.identifier as CFString, 1, nil) else {
            throw CocoaError(.fileWriteUnknown)
        }
        CGImageDestinationAddImage(destination, image, properties.map { $0 as CFDictionary })
        guard CGImageDestinationFinalize(destination) else { throw CocoaError(.fileWriteUnknown) }
        return file
    }

    private static func context(width: Int, height: Int) -> CGContext {
        CGContext(data: nil, width: width, height: height, bitsPerComponent: 8, bytesPerRow: 0,
                  space: CGColorSpace(name: CGColorSpace.sRGB)!, bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue)!
    }
}

/// Tamaño en píxeles de una imagen en disco.
func pixelSize(_ url: URL) -> (width: Int, height: Int)? {
    guard let source = CGImageSourceCreateWithURL(url as CFURL, nil),
          let properties = CGImageSourceCopyPropertiesAtIndex(source, 0, nil) as? [CFString: Any],
          let width = properties[kCGImagePropertyPixelWidth] as? Int,
          let height = properties[kCGImagePropertyPixelHeight] as? Int else { return nil }
    return (width, height)
}

/// Tipo (UTI) de una imagen en disco.
func imageType(_ url: URL) -> String? {
    guard let source = CGImageSourceCreateWithURL(url as CFURL, nil) else { return nil }
    return CGImageSourceGetType(source) as String?
}
```

```swift
// FILE: Sources/PoxiDropCore/ShelfItem.swift
import Foundation
import UniformTypeIdentifiers

public enum ItemKind: String, Codable, Sendable {
    case file, folder, image, text, link
}

/// Un elemento de una estantería.
public struct ShelfItem: Codable, Identifiable, Hashable, Sendable {
    public let id: UUID
    public var url: URL
    /// Marcador para seguir encontrando el archivo aunque se mueva (solo archivos del usuario).
    public var bookmark: Data?
    /// `true` si el archivo es una copia propia de PoxiDrop (texto soltado, imágenes de webs, resultados de acciones).
    public var isOwned: Bool
    public var kind: ItemKind
    public var addedAt: Date

    public init(id: UUID = UUID(), url: URL, bookmark: Data? = nil, isOwned: Bool, kind: ItemKind, addedAt: Date = Date()) {
        self.id = id
        self.url = url
        self.bookmark = bookmark
        self.isOwned = isOwned
        self.kind = kind
        self.addedAt = addedAt
    }

    public var name: String { url.lastPathComponent }
    public var exists: Bool { FileManager.default.fileExists(atPath: url.path) }

    /// Tipo de elemento según lo que hay en disco y la extensión.
    public static func kind(of url: URL) -> ItemKind {
        let values = try? url.resourceValues(forKeys: [.isDirectoryKey, .isPackageKey])
        if values?.isDirectory == true && values?.isPackage != true { return .folder }
        let ext = url.pathExtension.lowercased()
        if ext == "webloc" { return .link }
        guard let type = UTType(filenameExtension: ext) else { return .file }
        if type.conforms(to: .image) { return .image }
        if type.conforms(to: .plainText) { return .text }
        return .file
    }
}
```

```swift
import Foundation

public enum ShelfNaming {
    /// Nombre automático: el del archivo si hay uno solo, o «5 imágenes», «3 carpetas»…
    public static func automaticName(for items: [ShelfItem]) -> String {
        guard !items.isEmpty else { return "Estantería vacía" }
        if items.count == 1 { return items[0].name }
        return countLabel(for: items)
    }

    /// Texto de la píldora: «1 imagen», «5 imágenes», «2 elementos»…
    public static func countLabel(for items: [ShelfItem]) -> String {
        let count = items.count
        let kinds = Set(items.map(\.kind))
        let words: (singular: String, plural: String)
        if kinds.count == 1, let kind = kinds.first {
            switch kind {
            case .image: words = ("imagen", "imágenes")
            case .folder: words = ("carpeta", "carpetas")
            case .text: words = ("texto", "textos")
            case .link: words = ("enlace", "enlaces")
            case .file: words = ("archivo", "archivos")
            }
        } else {
            words = ("elemento", "elementos")
        }
        return "\(count) \(count == 1 ? words.singular : words.plural)"
    }
}
```

```swift
import Foundation

public enum FileSizes {
    /// Tamaño de un archivo, o la suma de todo lo que hay dentro si es una carpeta.
    public static func size(of url: URL) -> Int64 {
        guard let values = try? url.resourceValues(forKeys: [.isDirectoryKey, .fileSizeKey]) else { return 0 }
        guard values.isDirectory == true else { return Int64(values.fileSize ?? 0) }
        var total: Int64 = 0
        let enumerator = FileManager.default.enumerator(at: url, includingPropertiesForKeys: [.fileSizeKey])
        while let file = enumerator?.nextObject() as? URL {
            total += Int64((try? file.resourceValues(forKeys: [.fileSizeKey]).fileSize) ?? 0)
        }
        return total
    }

    public static func total(of urls: [URL]) -> Int64 {
        urls.reduce(0) { $0 + size(of: $1) }
    }

    /// «3,1 MB», en el idioma del sistema.
    public static func format(_ bytes: Int64) -> String {
        ByteCountFormatter.string(fromByteCount: bytes, countStyle: .file)
    }
}
```

```swift
// FILE: Sources/PoxiDropCore/ShelfStore.swift
import Foundation
import Combine
import CoreGraphics

public enum OwnedFolders {
    /// Crea una carpeta nueva y vacía dentro de `root` para guardar archivos propios.
    public static func make(in root: URL) throws -> URL {
        let folder = root.appendingPathComponent(UUID().uuidString, isDirectory: true)
        try FileManager.default.createDirectory(at: folder, withIntermediateDirectories: true)
        return folder
    }
}

/// Guarda las estanterías abiertas y las recientes, y los archivos propios de PoxiDrop.
@MainActor
public final class ShelfStore: ObservableObject {
    public static let recentLimit = 10

    @Published public private(set) var shelves: [Shelf] = []
    @Published public private(set) var recents: [Shelf] = []

    public let directory: URL
    public var itemsRoot: URL { directory.appendingPathComponent("Items", isDirectory: true) }
    private var stateURL: URL { directory.appendingPathComponent("estanterias.json") }

    private struct State: Codable {
        var shelves: [Shelf]
        var recents: [Shelf]
    }

    public static var defaultDirectory: URL {
        FileManager.default.urls(for: .applicationSupportDirectory, in: .userDomainMask)[0]
            .appendingPathComponent("PoxiDrop", isDirectory: true)
    }

    public init(directory: URL) {
        try? FileManager.default.createDirectory(at: directory, withIntermediateDirectories: true)
        self.directory = directory.resolvingSymlinksInPath()
        try? FileManager.default.createDirectory(at: itemsRoot, withIntermediateDirectories: true)
        load()
    }

    // MARK: Consultas

    public func shelf(_ id: Shelf.ID) -> Shelf? {
        shelves.first { $0.id == id }
    }

    public var favorite: Shelf? {
        shelves.first(where: \.isFavorite)
    }

    // MARK: Cambios

    @discardableResult
    public func createShelf(origin: CGPoint? = nil) -> Shelf.ID {
        let shelf = Shelf(origin: origin)
        shelves.append(shelf)
        save()
        return shelf.id
    }

    public func add(_ items: [ShelfItem], to id: Shelf.ID) {
        guard !items.isEmpty else { return }
        update(id) { shelf in
            var known = Set(shelf.items.map { $0.url.standardizedFileURL })
            for item in items where !known.contains(item.url.standardizedFileURL) {
                shelf.items.append(item)
                known.insert(item.url.standardizedFileURL)
            }
        }
    }

    /// Quita elementos. Con `deleteOwnedFiles` borra también las copias propias de PoxiDrop (nunca archivos del usuario).
    public func remove(_ itemIDs: Set<UUID>, from id: Shelf.ID, deleteOwnedFiles: Bool = true) {
        guard let shelf = shelf(id) else { return }
        if deleteOwnedFiles {
            deleteOwned(shelf.items.filter { itemIDs.contains($0.id) })
        }
        update(id) { $0.items.removeAll { itemIDs.contains($0.id) } }
    }

    public func clear(_ id: Shelf.ID) {
        guard let shelf = shelf(id) else { return }
        remove(Set(shelf.items.map(\.id)), from: id)
    }

    /// Cierra una estantería. La favorita se conserva; las demás pasan a recientes si tienen algo.
    public func close(_ id: Shelf.ID) {
        guard let index = shelves.firstIndex(where: { $0.id == id }) else { return }
        guard !shelves[index].isFavorite else { return }
        var closed = shelves.remove(at: index)
        if !closed.items.isEmpty {
            closed.origin = nil
            closed.dock = nil
            recents.insert(closed, at: 0)
            while recents.count > Self.recentLimit {
                deleteOwned(recents.removeLast().items)
            }
        }
        save()
    }

    @discardableResult
    public func reopen(recent id: Shelf.ID) -> Shelf.ID? {
        guard let index = recents.firstIndex(where: { $0.id == id }) else { return nil }
        let shelf = recents.remove(at: index)
        shelves.append(shelf)
        save()
        return shelf.id
    }

    public func clearRecents() {
        recents.forEach { deleteOwned($0.items) }
        recents = []
        save()
    }

    /// Solo puede haber una favorita.
    public func setFavorite(_ id: Shelf.ID, _ favorite: Bool) {
        for index in shelves.indices {
            if shelves[index].id == id {
                shelves[index].isFavorite = favorite
            } else if favorite {
                shelves[index].isFavorite = false
            }
        }
        save()
    }

    public func rename(_ id: Shelf.ID, to name: String?) {
        let trimmed = name?.trimmingCharacters(in: .whitespacesAndNewlines) ?? ""
        update(id) { $0.customName = trimmed.isEmpty ? nil : trimmed }
    }

    public func setOrigin(_ id: Shelf.ID, _ origin: CGPoint?) {
        update(id) { $0.origin = origin }
    }

    public func setDock(_ id: Shelf.ID, _ dock: DockEdge?) {
        update(id) { $0.dock = dock }
    }

    /// Tras renombrar o mover un archivo desde la app.
    public func replaceURL(of itemID: UUID, in id: Shelf.ID, with url: URL) {
        update(id) { shelf in
            guard let index = shelf.items.firstIndex(where: { $0.id == itemID }) else { return }
            shelf.items[index].url = url
            if !shelf.items[index].isOwned {
                shelf.items[index].bookmark = try? url.bookmarkData()
            }
        }
    }

    /// Carpeta nueva para guardar archivos propios (texto soltado, resultados de acciones).
    public func makeOwnedFolder() throws -> URL {
        try OwnedFolders.make(in: itemsRoot)
    }

    // MARK: Guardado

    public func save() {
        guard let data = try? JSONEncoder().encode(State(shelves: shelves, recents: recents)) else { return }
        try? data.write(to: stateURL, options: .atomic)
    }

    private func load() {
        guard let data = try? Data(contentsOf: stateURL),
              let state = try? JSONDecoder().decode(State.self, from: data) else { return }
        shelves = state.shelves.map(Self.resolvingBookmarks)
        recents = state.recents.map(Self.resolvingBookmarks)
    }

    private static func resolvingBookmarks(_ shelf: Shelf) -> Shelf {
        var shelf = shelf
        for index in shelf.items.indices {
            guard let bookmark = shelf.items[index].bookmark else { continue }
            var stale = false
            guard let url = try? URL(resolvingBookmarkData: bookmark, options: [.withoutUI], bookmarkDataIsStale: &stale) else { continue }
            shelf.items[index].url = url
            if stale { shelf.items[index].bookmark = try? url.bookmarkData() }
        }
        return shelf
    }

    private func update(_ id: Shelf.ID, _ change: (inout Shelf) -> Void) {
        guard let index = shelves.firstIndex(where: { $0.id == id }) else { return }
        change(&shelves[index])
        save()
    }

    /// Borra copias propias. Solo toca archivos dentro de `itemsRoot`.
    private func deleteOwned(_ items: [ShelfItem]) {
        let root = itemsRoot.resolvingSymlinksInPath().path + "/"
        for item in items where item.isOwned {
            let file = item.url.resolvingSymlinksInPath()
            guard file.path.hasPrefix(root) else { continue }
            try? FileManager.default.removeItem(at: file)
            let folder = file.deletingLastPathComponent()
            if folder.path + "/" != root,
               (try? FileManager.default.contentsOfDirectory(atPath: folder.path))?.isEmpty == true {
                try? FileManager.default.removeItem(at: folder)
            }
        }
    }
}
```

```swift
// FILE: Sources/PoxiDropCore/DropImporter.swift
import Foundation

/// Lo que puede llegar a una estantería.
public enum DropPayload: Sendable {
    /// Archivos o carpetas del usuario: se guardan como referencia, sin copiarlos.
    case files([URL])
    /// Archivos que ya están en la carpeta de PoxiDrop (promesas de Fotos o Mail, resultados de acciones).
    case ownedFiles([URL])
    /// Imagen sin archivo (por ejemplo, arrastrada desde una web).
    case image(Data, fileExtension: String)
    case text(String)
    case webLink(URL)
}

/// Convierte lo soltado en elementos de estantería. Lo que no es un archivo se guarda como archivo propio.
public struct DropImporter: Sendable {
    public let itemsRoot: URL

    public init(itemsRoot: URL) {
        self.itemsRoot = itemsRoot
    }

    public func items(from payload: DropPayload, now: Date = Date()) throws -> [ShelfItem] {
        switch payload {
        case .files(let urls):
            return urls.map { ShelfItem(url: $0, bookmark: try? $0.bookmarkData(), isOwned: false, kind: ShelfItem.kind(of: $0)) }
        case .ownedFiles(let urls):
            return urls.map { ShelfItem(url: $0, isOwned: true, kind: ShelfItem.kind(of: $0)) }
        case .image(let data, let fileExtension):
            let file = try OwnedFolders.make(in: itemsRoot).appendingPathComponent("Imagen \(Self.stamp(now)).\(fileExtension)")
            try data.write(to: file)
            return [ShelfItem(url: file, isOwned: true, kind: .image)]
        case .text(let text):
            let file = try OwnedFolders.make(in: itemsRoot).appendingPathComponent(Self.fileName(forText: text) + ".txt")
            try text.write(to: file, atomically: true, encoding: .utf8)
            return [ShelfItem(url: file, isOwned: true, kind: .text)]
        case .webLink(let link):
            let name = Self.fileName(forText: link.host() ?? "Enlace")
            let file = try OwnedFolders.make(in: itemsRoot).appendingPathComponent(name + ".webloc")
            let data = try PropertyListSerialization.data(fromPropertyList: ["URL": link.absoluteString], format: .xml, options: 0)
            try data.write(to: file)
            return [ShelfItem(url: file, isOwned: true, kind: .link)]
        }
    }

    /// Nombre de archivo a partir de la primera línea de un texto.
    static func fileName(forText text: String) -> String {
        let firstLine = text.split(whereSeparator: \.isNewline).first.map(String.init) ?? ""
        var cleaned = firstLine
            .replacingOccurrences(of: "/", with: "-")
            .replacingOccurrences(of: ":", with: "-")
            .trimmingCharacters(in: .whitespaces)
        while cleaned.hasPrefix(".") { cleaned.removeFirst() }
        let short = String(cleaned.prefix(40)).trimmingCharacters(in: .whitespaces)
        return short.isEmpty ? "Texto" : short
    }

    static func stamp(_ date: Date) -> String {
        let formatter = DateFormatter()
        formatter.locale = Locale(identifier: "es_ES")
        formatter.dateFormat = "yyyy-MM-dd HH.mm.ss"
        return formatter.string(from: date)
    }
}
```

```swift
import Foundation
import UniformTypeIdentifiers

public enum ActionGroup: String, CaseIterable, Sendable {
    case file = "Archivo"
    case image = "Imagen"
    case text = "Texto"
    case custom = "Mis acciones"
}

public struct ActionResult: Sendable {
    /// Archivos nuevos que se añaden a la estantería.
    public var files: [URL]
    /// Texto que la app copia al portapapeles (OCR).
    public var copiedText: String?

    public init(files: [URL] = [], copiedText: String? = nil) {
        self.files = files
        self.copiedText = copiedText
    }
}

public struct ActionError: LocalizedError, Equatable {
    public let message: String
    public init(_ message: String) { self.message = message }
    public var errorDescription: String? { message }
}

/// Una acción sobre los archivos de una estantería. Nunca modifica los originales:
/// los resultados se crean en `output`.
public protocol ShelfAction: Sendable {
    var id: String { get }
    var title: String { get }
    var symbol: String { get }
    var group: ActionGroup { get }
    func accepts(_ urls: [URL]) -> Bool
    func perform(_ urls: [URL], output: URL) async throws -> ActionResult
}

extension URL {
    var isImageFile: Bool {
        UTType(filenameExtension: pathExtension.lowercased())?.conforms(to: .image) == true
    }
}

enum UniqueName {
    /// Ruta libre en `folder`: «foto.png», «foto 2.png», «foto 3.png»…
    static func url(for name: String, in folder: URL) -> URL {
        let base = (name as NSString).deletingPathExtension
        let ext = (name as NSString).pathExtension
        var candidate = folder.appendingPathComponent(name)
        var number = 2
        while FileManager.default.fileExists(atPath: candidate.path) {
            candidate = folder.appendingPathComponent(ext.isEmpty ? "\(base) \(number)" : "\(base) \(number).\(ext)")
            number += 1
        }
        return candidate
    }
}

enum ProcessRunner {
    /// Ejecuta un programa y devuelve su salida. Si falla, lanza `ActionError` con lo que escribió en stderr.
    @discardableResult
    static func run(_ executable: String, _ arguments: [String], environment: [String: String] = [:]) throws -> String {
        let process = Process()
        process.executableURL = URL(fileURLWithPath: executable)
        process.arguments = arguments
        if !environment.isEmpty {
            process.environment = ProcessInfo.processInfo.environment.merging(environment) { $1 }
        }
        let errorFile = FileManager.default.temporaryDirectory.appendingPathComponent("poxidrop-\(UUID().uuidString).log")
        FileManager.default.createFile(atPath: errorFile.path, contents: nil)
        defer { try? FileManager.default.removeItem(at: errorFile) }
        let errorHandle = try FileHandle(forWritingTo: errorFile)
        let output = Pipe()
        process.standardOutput = output
        process.standardError = errorHandle
        try process.run()
        let data = output.fileHandleForReading.readDataToEndOfFile()
        process.waitUntilExit()
        try? errorHandle.close()
        guard process.terminationStatus == 0 else {
            let message = (try? String(contentsOf: errorFile, encoding: .utf8))?
                .trimmingCharacters(in: .whitespacesAndNewlines) ?? ""
            throw ActionError(message.isEmpty ? "El proceso terminó con el código \(process.terminationStatus)." : message)
        }
        return String(data: data, encoding: .utf8) ?? ""
    }
}
```

```swift
import Foundation

public struct ZipAction: ShelfAction {
    public let id = "file.zip"
    public let title = "Comprimir en ZIP"
    public let symbol = "doc.zipper"
    public let group = ActionGroup.file

    public init() {}

    public func accepts(_ urls: [URL]) -> Bool { !urls.isEmpty }

    public func perform(_ urls: [URL], output: URL) async throws -> ActionResult {
        let name = urls.count == 1 ? urls[0].deletingPathExtension().lastPathComponent : "Archivos"
        let zip = UniqueName.url(for: "\(name).zip", in: output)
        if urls.count == 1 {
            try ProcessRunner.run("/usr/bin/ditto", ["-c", "-k", "--sequesterRsrc", "--keepParent", urls[0].path, zip.path])
        } else {
            let staging = FileManager.default.temporaryDirectory.appendingPathComponent("poxidrop-zip-\(UUID().uuidString)", isDirectory: true)
            try FileManager.default.createDirectory(at: staging, withIntermediateDirectories: true)
            defer { try? FileManager.default.removeItem(at: staging) }
            for url in urls {
                try FileManager.default.copyItem(at: url, to: UniqueName.url(for: url.lastPathComponent, in: staging))
            }
            try ProcessRunner.run("/usr/bin/ditto", ["-c", "-k", "--sequesterRsrc", staging.path, zip.path])
        }
        return ActionResult(files: [zip])
    }
}

public enum FileOps {
    /// Copia a una carpeta sin pisar nada. Devuelve las rutas nuevas.
    public static func copy(_ urls: [URL], to folder: URL) throws -> [URL] {
        try urls.map { url in
            let target = UniqueName.url(for: url.lastPathComponent, in: folder)
            try FileManager.default.copyItem(at: url, to: target)
            return target
        }
    }

    /// Mueve a una carpeta sin pisar nada. Devuelve las rutas nuevas.
    public static func move(_ urls: [URL], to folder: URL) throws -> [URL] {
        try urls.map { url in
            let target = UniqueName.url(for: url.lastPathComponent, in: folder)
            try FileManager.default.moveItem(at: url, to: target)
            return target
        }
    }

    /// Cambia el nombre de un archivo en su misma carpeta.
    public static func rename(_ url: URL, to newName: String) throws -> URL {
        let name = newName.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !name.isEmpty else { throw ActionError("Escribe un nombre.") }
        guard !name.contains("/"), !name.contains(":") else { throw ActionError("El nombre no puede llevar «/» ni «:».") }
        guard name.utf8.count <= 255 else { throw ActionError("El nombre es demasiado largo.") }
        let target = url.deletingLastPathComponent().appendingPathComponent(name)
        if target.path == url.path { return url }
        guard !FileManager.default.fileExists(atPath: target.path) else {
            throw ActionError("Ya existe un archivo llamado «\(name)».")
        }
        try FileManager.default.moveItem(at: url, to: target)
        return target
    }
}
```

```swift
// FILE: Sources/PoxiDropCore/Actions/ImageActions.swift
import Foundation
import CoreGraphics
import ImageIO
import UniformTypeIdentifiers

enum ImageIOHelper {
    /// Carga una imagen ya girada según su orientación EXIF.
    static func load(_ url: URL) throws -> CGImage {
        guard let source = CGImageSourceCreateWithURL(url as CFURL, nil),
              let properties = CGImageSourceCopyPropertiesAtIndex(source, 0, nil) as? [CFString: Any],
              let width = properties[kCGImagePropertyPixelWidth] as? Int,
              let height = properties[kCGImagePropertyPixelHeight] as? Int else {
            throw ActionError("No se puede leer la imagen «\(url.lastPathComponent)».")
        }
        let options: [CFString: Any] = [
            kCGImageSourceCreateThumbnailFromImageAlways: true,
            kCGImageSourceCreateThumbnailWithTransform: true,
            kCGImageSourceThumbnailMaxPixelSize: max(width, height),
            kCGImageSourceShouldCacheImmediately: true,
        ]
        guard let image = CGImageSourceCreateThumbnailAtIndex(source, 0, options as CFDictionary) else {
            throw ActionError("No se puede leer la imagen «\(url.lastPathComponent)».")
        }
        return image
    }

    static func context(width: Int, height: Int) throws -> CGContext {
        guard let context = CGContext(data: nil, width: width, height: height, bitsPerComponent: 8, bytesPerRow: 0,
                                      space: CGColorSpace(name: CGColorSpace.sRGB)!,
                                      bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue) else {
            throw ActionError("La imagen es demasiado grande.")
        }
        context.interpolationQuality = .high
        return context
    }

    static func scaled(_ image: CGImage, width: Int, height: Int) throws -> CGImage {
        let context = try context(width: width, height: height)
        context.draw(image, in: CGRect(x: 0, y: 0, width: width, height: height))
        guard let result = context.makeImage() else { throw ActionError("No se pudo redimensionar la imagen.") }
        return result
    }

    /// Guarda una imagen sin metadatos. Para JPEG rellena la transparencia de blanco.
    static func write(_ image: CGImage, to url: URL, type: UTType, quality: Double = 0.9) throws {
        var output = image
        if type == .jpeg {
            let context = try context(width: image.width, height: image.height)
            context.setFillColor(CGColor(gray: 1, alpha: 1))
            context.fill(CGRect(x: 0, y: 0, width: image.width, height: image.height))
            context.draw(image, in: CGRect(x: 0, y: 0, width: image.width, height: image.height))
            output = context.makeImage() ?? image
        }
        guard let destination = CGImageDestinationCreateWithURL(url as CFURL, type.identifier as CFString, 1, nil) else {
            throw ActionError("Este Mac no puede guardar imágenes en formato \(fileExtension(for: type).uppercased()).")
        }
        let properties: [CFString: Any] = [kCGImageDestinationLossyCompressionQuality: quality]
        CGImageDestinationAddImage(destination, output, properties as CFDictionary)
        guard CGImageDestinationFinalize(destination) else {
            throw ActionError("No se pudo guardar «\(url.lastPathComponent)».")
        }
    }

    /// El tipo del original si se puede escribir; si no, PNG.
    static func writableType(for url: URL) -> UTType {
        let type = UTType(filenameExtension: url.pathExtension.lowercased()) ?? .png
        let writable = CGImageDestinationCopyTypeIdentifiers() as? [String] ?? []
        return writable.contains(type.identifier) ? type : .png
    }

    static func fileExtension(for type: UTType) -> String {
        type == .jpeg ? "jpg" : (type.preferredFilenameExtension ?? "png")
    }

    /// «foto (50 %).png» en la carpeta de salida, sin pisar nada.
    static func outputURL(for source: URL, suffix: String?, type: UTType, in folder: URL) -> URL {
        let base = source.deletingPathExtension().lastPathComponent
        let name = suffix.map { "\(base) (\($0))" } ?? base
        return UniqueName.url(for: "\(name).\(fileExtension(for: type))", in: folder)
    }
}

public struct ResizeImagesAction: ShelfAction {
    public enum Mode: Sendable, Equatable {
        case percent(Int)
        case maxSide(Int)
    }

    public let mode: Mode
    public init(mode: Mode) { self.mode = mode }

    public var id: String {
        switch mode {
        case .percent(let percent): return "image.resize.percent.\(percent)"
        case .maxSide(let side): return "image.resize.side.\(side)"
        }
    }

    public var title: String {
        switch mode {
        case .percent(let percent): return "Redimensionar al \(percent) %"
        case .maxSide(let side): return "Redimensionar a \(side) px"

    public func perform(_ urls: [URL], output: URL) async throws -> ActionResult {
        var files: [URL] = []
        for url in urls {
            let image = try ImageIOHelper.load(url)
            let size = targetSize(width: image.width, height: image.height)

        }
        return ActionResult(files: files)
    }

        switch mode {
        case .percent(let percent): return "\(percent) %"
        case .maxSide(let side): return "\(side) px"
        }

        switch mode {
        case .percent(let percent):
            return (max(1, width * percent / 100), max(1, height * percent / 100))
        case .maxSide(let side):
            let scale = min(1, Double(side) / Double(max(width, height)))
            return (max(1, Int((Double(width) * scale).rounded())), max(1, Int((Double(height) * scale).rounded())))
        }
    }
```

```swift
import Foundation
import AppKit
import PDFKit
import Vision

public struct ExtractTextAction: ShelfAction {
    public init() {}

    public let id = "image.ocr"
    public let title = "Extraer texto (OCR)"
    public let symbol = "text.viewfinder"
    public let group = ActionGroup.image

    public func accepts(_ urls: [URL]) -> Bool { !urls.isEmpty && urls.allSatisfy(\.isImageFile) }

    public func perform(_ urls: [URL], output: URL) async throws -> ActionResult {
        var texts: [String] = []
        for url in urls {
            let request = VNRecognizeTextRequest()
            request.recognitionLevel = .accurate
            request.recognitionLanguages = ["es-ES", "en-US"]
            request.usesLanguageCorrection = true
            try VNImageRequestHandler(cgImage: try ImageIOHelper.load(url)).perform([request])
            let lines = (request.results ?? []).compactMap { $0.topCandidates(1).first?.string }
            if !lines.isEmpty { texts.append(lines.joined(separator: "\n")) }
        }
        guard !texts.isEmpty else { throw ActionError("No se ha encontrado texto en la imagen.") }
        let text = texts.joined(separator: "\n\n")
        let file = UniqueName.url(for: "Texto extraído.txt", in: output)
        try text.write(to: file, atomically: true, encoding: .utf8)
        return ActionResult(files: [file], copiedText: text)
    }
}

public struct MakePDFAction: ShelfAction {
    public init() {}

    public let id = "image.pdf"
    public let title = "Crear PDF"
    public let symbol = "doc.richtext"
    public let group = ActionGroup.image

    public func accepts(_ urls: [URL]) -> Bool { !urls.isEmpty && urls.allSatisfy(\.isImageFile) }

    public func perform(_ urls: [URL], output: URL) async throws -> ActionResult {
        let document = PDFDocument()
        for (index, url) in urls.enumerated() {
            guard let image = NSImage(contentsOf: url), let page = PDFPage(image: image) else {
                throw ActionError("No se puede añadir «\(url.lastPathComponent)» al PDF.")
            }
            document.insert(page, at: index)
        }
        let name = urls.count == 1 ? urls[0].deletingPathExtension().lastPathComponent + ".pdf" : "Imágenes.pdf"
        let file = UniqueName.url(for: name, in: output)
        guard document.write(to: file) else { throw ActionError("No se pudo crear el PDF.") }
        return ActionResult(files: [file])
    }
}

public struct PlainTextAction: ShelfAction {
    static let extensions: Set<String> = ["rtf", "rtfd", "html", "htm", "doc", "docx", "odt", "webarchive"]

    public init() {}

    public let id = "text.plain"
    public let title = "Convertir a texto plano"
    public let symbol = "textformat"
    public let group = ActionGroup.text

    public func accepts(_ urls: [URL]) -> Bool {
        !urls.isEmpty && urls.allSatisfy { Self.extensions.contains($0.pathExtension.lowercased()) }
    }

    public func perform(_ urls: [URL], output: URL) async throws -> ActionResult {
        var files: [URL] = []
        for url in urls {
            // Leer HTML usa WebKit, que solo funciona en el hilo principal.
            let text = try await MainActor.run {
                try NSAttributedString(url: url, options: [:], documentAttributes: nil).string
            }
            let file = UniqueName.url(for: url.deletingPathExtension().lastPathComponent + ".txt", in: output)
            try text.write(to: file, atomically: true, encoding: .utf8)
            files.append(file)
        }
        return ActionResult(files: files)
    }
}

/// Ejecuta un script con las rutas como argumentos. Lo que deje en `$POXIDROP_OUTPUT` vuelve a la estantería.
public struct ScriptAction: ShelfAction {
    public let script: CustomScript
    public init(script: CustomScript) { self.script = script }

    public var id: String { "custom.\(script.id.uuidString)" }
    public var title: String { script.name }
    public var symbol: String { "terminal" }
    public var group: ActionGroup { .custom }

    public func accepts(_ urls: [URL]) -> Bool { !urls.isEmpty }

    public func perform(_ urls: [URL], output: URL) async throws -> ActionResult {
        let paths = urls.map(\.path)
        let environment = ["POXIDROP_OUTPUT": output.path]
        if FileManager.default.isExecutableFile(atPath: script.path) {
            try ProcessRunner.run(script.path, paths, environment: environment)
        } else {
            try ProcessRunner.run("/bin/zsh", [script.path] + paths, environment: environment)
        }
        let produced = (try? FileManager.default.contentsOfDirectory(at: output, includingPropertiesForKeys: nil)) ?? []
        return ActionResult(files: produced.sorted { $0.lastPathComponent < $1.lastPathComponent })
    }
}
```

```swift
        PlainTextAction(),
    ]

    /// Acciones que se pueden hacer con estos archivos.
    public static func available(for urls: [URL], custom: [any ShelfAction] = []) -> [any ShelfAction] {
        (builtIn + custom).filter { $0.accepts(urls) }
    }

    /// Filtra por título sin tener en cuenta mayúsculas ni acentos.
    public static func search(_ query: String, in actions: [any ShelfAction]) -> [any ShelfAction] {
        let needle = normalized(query)
        guard !needle.isEmpty else { return actions }
        return actions.filter { normalized($0.title).contains(needle) }
    }

    public static func normalized(_ text: String) -> String {
        text.folding(options: [.caseInsensitive, .diacriticInsensitive], locale: Locale(identifier: "es_ES"))
            .trimmingCharacters(in: .whitespaces)
    }
```

```swift
import AppKit
import Carbon.HIToolbox

struct HotKey {
    let keyCode: UInt32
    let modifiers: UInt32
    let action: () -> Void
}

/// Atajos de teclado globales con Carbon (no necesita el permiso de Accesibilidad).
@MainActor
final class HotKeys {
    private var references: [EventHotKeyRef] = []
    private var actions: [UInt32: () -> Void] = [:]
    private var handler: EventHandlerRef?

    func register(_ hotKeys: [HotKey]) {
        unregisterAll()
        installHandlerIfNeeded()
        for (index, hotKey) in hotKeys.enumerated() {
            let number = UInt32(index + 1)
            var reference: EventHotKeyRef?
            let identifier = EventHotKeyID(signature: OSType(0x5044_5250), id: number) // 'PDRP'
            let status = RegisterEventHotKey(hotKey.keyCode, hotKey.modifiers, identifier, GetApplicationEventTarget(), 0, &reference)
            guard status == noErr, let reference else { continue }
            references.append(reference)
            actions[number] = hotKey.action
        }
    }

    func unregisterAll() {
        references.forEach { UnregisterEventHotKey($0) }
        references = []
        actions = [:]
    }

    fileprivate func fire(_ number: UInt32) {
        actions[number]?()
    }

    private func installHandlerIfNeeded() {
        guard handler == nil else { return }
        var spec = EventTypeSpec(eventClass: OSType(kEventClassKeyboard), eventKind: UInt32(kEventHotKeyPressed))
        InstallEventHandler(GetApplicationEventTarget(), { _, event, userData in
            var identifier = EventHotKeyID()
            GetEventParameter(event, EventParamName(kEventParamDirectObject), EventParamType(typeEventHotKeyID),
                              nil, MemoryLayout<EventHotKeyID>.size, nil, &identifier)
            guard let userData else { return noErr }
            let hotKeys = Unmanaged<HotKeys>.fromOpaque(userData).takeUnretainedValue()
            MainActor.assumeIsolated { hotKeys.fire(identifier.id) }
            return noErr
        }, 1, &spec, Unmanaged.passUnretained(self).toOpaque(), &handler)
    }
}
```

```swift
import AppKit
import PoxiDropCore

enum PasteboardReader {
    static let dropTypes: [NSPasteboard.PasteboardType] =
        [.fileURL, .URL, .png, .tiff, .string] + NSFilePromiseReceiver.readableDraggedTypes.map { NSPasteboard.PasteboardType($0) }

    /// Lee archivos, imágenes, enlaces o texto (en ese orden de preferencia).
    static func payloads(from pasteboard: NSPasteboard) -> [DropPayload] {
        if let files = pasteboard.readObjects(forClasses: [NSURL.self], options: [.urlReadingFileURLsOnly: true]) as? [URL], !files.isEmpty {
            return [.files(files)]
        }
        if let png = pasteboard.data(forType: .png) {
            return [.image(png, fileExtension: "png")]
        }
        if let tiff = pasteboard.data(forType: .tiff),
           let png = NSBitmapImageRep(data: tiff)?.representation(using: .png, properties: [:]) {
            return [.image(png, fileExtension: "png")]
        }
        if let urls = pasteboard.readObjects(forClasses: [NSURL.self]) as? [URL],
           let link = urls.first(where: { $0.scheme == "http" || $0.scheme == "https" }) {
            return [.webLink(link)]
        }
        if let text = pasteboard.string(forType: .string), !text.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty {
            return [.text(text)]
        }
        return []
    }
}
```

```swift
/// Miniaturas de Quick Look con caché. Mientras se generan devuelve el icono del Finder.
@MainActor
final class Thumbnails: ObservableObject {
    static let shared = Thumbnails()

    @Published private(set) var images: [URL: NSImage] = [:]
    private var pending = Set<URL>()

    func image(for url: URL) -> NSImage {
        if let image = images[url] { return image }
        request(url)
        return NSWorkspace.shared.icon(forFile: url.path)
    }

    func cached(_ url: URL) -> NSImage? { images[url] }

    /// Carga una imagen al momento (para las capturas de prueba).
    func preload(_ url: URL) {
        if let image = NSImage(contentsOf: url) { images[url] = image }
    }

    private func request(_ url: URL) {
        guard !pending.contains(url) else { return }
        pending.insert(url)
        let request = QLThumbnailGenerator.Request(fileAt: url, size: CGSize(width: 160, height: 160),
                                                   scale: NSScreen.main?.backingScaleFactor ?? 2, representationTypes: .all)
        QLThumbnailGenerator.shared.generateBestRepresentation(for: request) { representation, _ in
            guard let image = representation?.nsImage else { return }
            DispatchQueue.main.async {
                MainActor.assumeIsolated { Thumbnails.shared.images[url] = image }
            }
        }
    }
}

/// Tamaños de archivos y carpetas calculados en segundo plano.
@MainActor
final class SizeCache: ObservableObject {
    static let shared = SizeCache()

    @Published private(set) var sizes: [URL: Int64] = [:]
    private var pending = Set<URL>()

    func size(for url: URL) -> Int64? {
        if let size = sizes[url] { return size }
        guard !pending.contains(url) else { return nil }
        pending.insert(url)
        Task.detached(priority: .utility) {
            let size = FileSizes.size(of: url)
            await MainActor.run { SizeCache.shared.sizes[url] = size }
        }
        return nil
    }

    /// Suma de todos, o `nil` si alguno aún se está calculando.
    func total(for urls: [URL]) -> Int64? {
        var total: Int64 = 0
```

```swift
        NSScreen.screens.contains { rect(for: $0) != nil }
    }

    /// Zona de la muesca en coordenadas de pantalla, algo más ancha para que sea fácil de alcanzar.
    static func rect(for screen: NSScreen) -> CGRect? {
        let top = screen.safeAreaInsets.top
        guard top > 0, let left = screen.auxiliaryTopLeftArea, let right = screen.auxiliaryTopRightArea else { return nil }
        let width = screen.frame.width - left.width - right.width
        guard width > 0 else { return nil }
        return CGRect(x: screen.frame.minX + left.width, y: screen.frame.maxY - top, width: width, height: top + 1)
```

```swift
            let draggingItem = NSDraggingItem(pasteboardWriter: item.url as NSURL)

            draggingItem.setDraggingFrame(NSRect(x: point.x - 32 + offset, y: point.y - 32 - offset, width: 64, height: 64), contents: image)
            return draggingItem
        }
        let session = beginDraggingSession(with: draggingItems, event: event, source: self)
        session.animatesToStartingPositionsOnCancelOrFail = true
        session.draggingFormation = .pile
    }

    override func mouseUp(with event: NSEvent) {

        menuProvider?()

        context == .withinApplication ? .copy : operation()
```

**Files:**

**Interfaces:**
- Consumes: `ShakeDetector`, `AppSettings`, `NotchZone`.
- Produces: `DragWatcher(settings:)` con `enum Trigger { case shake, modifier, notch }`, `onTrigger: ((CGPoint, Trigger) -> Void)?`, `onDragEnded: (() -> Void)?`, `start()`.

Funcionamiento: un monitor global de `leftMouseDown`/`leftMouseDragged` (no requiere permisos) arranca un temporizador a 60 Hz mientras el botón está pulsado. El arrastre «con contenido» se reconoce porque cambia el `changeCount` del portapapeles de arrastre. Se lee la posición con `NSEvent.mouseLocation` y las teclas con `NSEvent.modifierFlags`, sin depender de que macOS entregue eventos durante la sesión de arrastre.

- [ ] **Step 1: Escribir el archivo**

```swift
// FILE: Sources/PoxiDrop/DragWatcher.swift
import AppKit
import PoxiDropCore

/// Vigila los arrastres de todo el sistema y avisa cuando hay que abrir una estantería.
@MainActor
final class DragWatcher {
    enum Trigger {
        case shake, modifier, notch
    }

    var onTrigger: ((CGPoint, Trigger) -> Void)?
    var onDragEnded: (() -> Void)?

    private let settings: AppSettings
    private var monitor: Any?
    private var timer: Timer?
    private var detector = ShakeDetector()
    private var baselineChangeCount = 0
    private var dragging = false
    private var triggered = false

    init(settings: AppSettings) {
        self.settings = settings
    }

    func start() {
        guard monitor == nil else { return }
        monitor = NSEvent.addGlobalMonitorForEvents(matching: [.leftMouseDown, .leftMouseDragged]) { [weak self] _ in
            MainActor.assumeIsolated { self?.startPolling() }
        }
    }

    private func startPolling() {
        guard timer == nil else { return }
        baselineChangeCount = NSPasteboard(name: .drag).changeCount
        detector = ShakeDetector(settings: .sensitivity(settings.shakeSensitivity))
        dragging = false
        triggered = false
        let timer = Timer(timeInterval: 1.0 / 60, repeats: true) { [weak self] _ in
            MainActor.assumeIsolated { self?.tick() }
        }
        RunLoop.main.add(timer, forMode: .common)
        self.timer = timer
    }

    private func tick() {
        guard NSEvent.pressedMouseButtons & 1 == 1 else {
            stop()
            return
        }
        if !dragging {
            guard NSPasteboard(name: .drag).changeCount != baselineChangeCount else { return }
            dragging = true
        }
        guard !triggered else { return }

        let point = NSEvent.mouseLocation
        let trigger: Trigger?
        if settings.modifierEnabled && NSEvent.modifierFlags.contains(settings.modifierKey.flag) {
            trigger = .modifier
        } else if settings.notchEnabled && NotchZone.contains(point) {
            trigger = .notch
        } else if settings.shakeEnabled && detector.add(x: point.x, time: ProcessInfo.processInfo.systemUptime) {
            trigger = .shake
        } else {
            trigger = nil
        }
        if let trigger {
            triggered = true
            onTrigger?(point, trigger)
        }
    }

    private func stop() {
        timer?.invalidate()
        timer = nil
        if dragging { onDragEnded?() }
        dragging = false
    }
}
```

```swift
    var onEnterExit: ((Bool) -> Void)?

        super.init(frame: frame)
        content.frame = bounds
        content.autoresizingMask = [.width, .height]
        addSubview(content)
        registerForDraggedTypes([.fileURL])
    }

    required init?(coder: NSCoder) { fatalError("init(coder:) no se usa") }

    /// Casilla bajo un punto (coordenadas de esta vista, origen abajo a la izquierda).
    func action(at point: CGPoint) -> QuickAction? {
        guard point.y < 196, point.x > 6, point.x < 190, point.y > 6, point.y < 190 else { return nil }
        let column = point.x < 98 ? 0 : 1
        let row = point.y >= 98 ? 0 : 1
        return QuickAction.allCases[row * 2 + column]
    }

    override func draggingEntered(_ sender: NSDraggingInfo) -> NSDragOperation {
        onEnterExit?(true)
        return draggingUpdated(sender)
    }

    override func draggingUpdated(_ sender: NSDraggingInfo) -> NSDragOperation {
        let action = action(at: convert(sender.draggingLocation, from: nil))
        model?.hovered = action
        return action == nil ? [] : .copy
    }

    override func draggingExited(_ sender: NSDraggingInfo?) {
        model?.hovered = nil
        onEnterExit?(false)
    }

    override func performDragOperation(_ sender: NSDraggingInfo) -> Bool {
        guard let action = action(at: convert(sender.draggingLocation, from: nil)),
              let urls = sender.draggingPasteboard.readObjects(forClasses: [NSURL.self], options: [.urlReadingFileURLsOnly: true]) as? [URL],
              !urls.isEmpty else { return false }
        onDrop?(action, urls)
        return true
    }
}

    func cancelHide() {
        hideTask?.cancel()
    }

        hideTask?.cancel()
        model.hovered = nil
```

```swift
struct PaletteEntry: Identifiable {

    let title: String
    let symbol: String
    let run: () -> Void

@MainActor
final class PaletteModel: ObservableObject {
    @Published var query = "" { didSet { highlighted = 0 } }
    @Published var highlighted = 0
    let entries: [PaletteEntry]

    init(entries: [PaletteEntry]) {
        self.entries = entries
    }

    var results: [PaletteEntry] {
        let needle = ActionRegistry.normalized(query)
        guard !needle.isEmpty else { return entries }
        return entries.filter { ActionRegistry.normalized($0.title).contains(needle) }
    }

    func move(_ delta: Int) {
        let count = results.count
        guard count > 0 else { return }
        highlighted = min(max(highlighted + delta, 0), count - 1)
    }
}
```

```swift
                model.compact = false
                panel.setFrame(CGRect(origin: panel.frame.origin, size: ShelfGeometry.shelfSize), display: true)
            }
            return
        }
        if !model.isDropTarget && !panel.frame.contains(NSEvent.mouseLocation) && !model.compact {

        }
    }

        let screen = NSScreen.screens.first { $0.frame.contains(panel.frame.center) } ?? NSScreen.main ?? NSScreen.screens[0]
        let area = screen.visibleFrame
        let y = min(max(panel.frame.midY - size.height / 2, area.minY), area.maxY - size.height)
        let x = model.shelf.dock == .left ? area.minX : area.maxX - size.width
        return CGRect(x: x, y: y, width: size.width, height: size.height)
    }

    func expand() {
        guard model.shelf.dock != nil, model.compact else { return }
        collapseTask?.cancel()
        model.compact = false
        panel.setFrame(dockedFrame(size: ShelfGeometry.shelfSize), display: true, animate: true)
    }

    func collapse() {
        guard model.shelf.dock != nil else { return }
        hideDetail()
        model.compact = true
        panel.setFrame(dockedFrame(size: ShelfGeometry.compactSize), display: true, animate: true)
    }

    private func scheduleCollapse() {
        guard model.shelf.dock != nil else { return }
        collapseTask?.cancel()
        collapseTask = Task { @MainActor [weak self] in
            try? await Task.sleep(nanoseconds: 700_000_000)
            guard let self, !Task.isCancelled else { return }
            let mouse = NSEvent.mouseLocation
            guard !self.panel.frame.contains(mouse), !(self.detailPanel?.frame.contains(mouse) ?? false) else { return }
            self.collapse()
        }
    }
```

```swift
// FILE: Sources/PoxiDrop/ShelfManager.swift
import AppKit
import Combine
import PoxiDropCore

/// Coordina las estanterías: sus ventanas, lo que reciben y las acciones.
@MainActor
final class ShelfManager {
    let store: ShelfStore
    let settings: AppSettings
    let openSettings: () -> Void
    let quickActions = QuickActionsController()

    private var controllers: [Shelf.ID: ShelfWindowController] = [:]
    private var subscriptions = Set<AnyCancellable>()
    private var dragShelf: Shelf.ID?
    private(set) var activeShelf: Shelf.ID? {
        didSet { updateActive() }
    }

    init(store: ShelfStore, settings: AppSettings, openSettings: @escaping () -> Void) {
        self.store = store
        self.settings = settings
        self.openSettings = openSettings
        store.$shelves
            .receive(on: RunLoop.main)
            .sink { [weak self] _ in MainActor.assumeIsolated { self?.sync() } }
            .store(in: &subscriptions)
        settings.$detailLayout
            .receive(on: RunLoop.main)
            .sink { [weak self] _ in MainActor.assumeIsolated { self?.controllers.values.forEach { $0.layoutDetail() } } }
            .store(in: &subscriptions)
    }

    func controller(for id: Shelf.ID) -> ShelfWindowController? {
        controllers[id]
    }

    /// Crea o quita ventanas para que coincidan con las estanterías guardadas.
    func sync() {
        let ids = Set(store.shelves.map(\.id))
        for (id, controller) in controllers where !ids.contains(id) {
            controller.close()
            controllers[id] = nil
        }
        for shelf in store.shelves {
            if let controller = controllers[shelf.id] {
                controller.refresh()
                continue
            }
            let controller = ShelfWindowController(shelf: shelf, manager: self)
            controllers[shelf.id] = controller
            controller.show(at: shelf.origin ?? ShelfGeometry.origin(centeredAt: defaultPoint, size: ShelfGeometry.shelfSize))
            controller.refresh()
        }
        updateActive()
    }

    private var defaultPoint: CGPoint {
        let area = (NSScreen.main ?? NSScreen.screens[0]).visibleFrame
        return CGPoint(x: area.maxX - 140, y: area.maxY - 140)
    }

    private func updateActive() {
        for (id, controller) in controllers {
            controller.model.isActive = id == activeShelf
        }
    }

    // MARK: Estanterías

    @discardableResult
    func newShelf(near point: CGPoint? = nil) -> Shelf.ID {
        let origin = ShelfGeometry.origin(centeredAt: point ?? NSEvent.mouseLocation, size: ShelfGeometry.shelfSize)
        let id = store.createShelf(origin: origin)
        activeShelf = id
        sync()
        return id
    }

    func activate(_ id: Shelf.ID) {
        activeShelf = id
    }

    func close(_ id: Shelf.ID) {
        if store.shelf(id)?.isFavorite == true {
            controllers[id]?.hide()
        }
        store.close(id)
        if activeShelf == id { activeShelf = nil }
        sync()
    }

    func closeAll() {
        for shelf in store.shelves { close(shelf.id) }
    }

    var anyVisible: Bool {
        controllers.values.contains { $0.isVisible }
    }

    func toggleAll() {
        let show = !anyVisible
        controllers.values.forEach { show ? $0.showAgain() : $0.hide() }
    }

    func showFavorite() {
        if let favorite = store.favorite {
            controllers[favorite.id]?.showAgain()
            activeShelf = favorite.id
        } else {
            let id = newShelf()
            store.setFavorite(id, true)
        }
    }

    func reopenRecent(_ id: Shelf.ID) {
        guard let reopened = store.reopen(recent: id) else { return }
        activeShelf = reopened
        sync()
    }

    // MARK: Arrastres

    /// El vigía pide una estantería durante un arrastre.
    func openShelfForDrag(at point: CGPoint, trigger: DragWatcher.Trigger) {
        var center = point
        if trigger == .notch, let screen = NSScreen.screens.first(where: { $0.frame.contains(point) }) {
            center = CGPoint(x: point.x, y: screen.frame.maxY - screen.safeAreaInsets.top - ShelfGeometry.shelfSize.height / 2 - 10)
        }
        dragShelf = newShelf(near: center)
    }

    /// Al terminar el arrastre: fuera las acciones directas, y si la estantería que se abrió se quedó vacía, se cierra.
    func dragSessionEnded() {
        quickActions.scheduleHide()
        guard let id = dragShelf else { return }
        dragShelf = nil
        Task { @MainActor [weak self] in
            try? await Task.sleep(nanoseconds: 1_500_000_000)
            guard let self, let shelf = self.store.shelf(id), shelf.items.isEmpty,
                  self.controllers[id]?.model.busy != true else { return }
            self.close(id)
        }
    }

    func showQuickActions(for controller: ShelfWindowController) {
        quickActions.perform = { [weak controller] action, urls in controller?.performQuickAction(action, urls: urls) }
        quickActions.show(below: controller.panel.frame)
    }

    // MARK: Contenido

    func addClipboardToNewShelf() {
        addFromClipboard(to: newShelf())
    }

    func addFromClipboard(to id: Shelf.ID) {
        let payloads = PasteboardReader.payloads(from: .general)
        guard !payloads.isEmpty else {
            controllers[id]?.model.show("El portapapeles está vacío", isError: true)
            return
        }
        importPayloads(payloads, into: id)
    }

    func importPayloads(_ payloads: [DropPayload], into id: Shelf.ID) {
        let importer = DropImporter(itemsRoot: store.itemsRoot)
        var items: [ShelfItem] = []
        for payload in payloads {
            do {
                items += try importer.items(from: payload)
            } catch {
                controllers[id]?.model.show(error.localizedDescription, isError: true)
            }
        }
        store.add(items, to: id)
        sync()
    }

    /// Lo que se suelta en una estantería: archivos, o promesas de archivo (Fotos, Mail…), o lo que haya.
    func receive(_ pasteboard: NSPasteboard, into id: Shelf.ID) {
        let files = pasteboard.readObjects(forClasses: [NSURL.self], options: [.urlReadingFileURLsOnly: true]) as? [URL] ?? []
        let promises = pasteboard.readObjects(forClasses: [NSFilePromiseReceiver.self]) as? [NSFilePromiseReceiver] ?? []
        if files.isEmpty, !promises.isEmpty {
            receivePromises(promises, into: id)
        } else {
            importPayloads(PasteboardReader.payloads(from: pasteboard), into: id)
        }
    }

    private func receivePromises(_ promises: [NSFilePromiseReceiver], into id: Shelf.ID) {
        guard let folder = try? OwnedFolders.make(in: store.itemsRoot) else { return }
        controllers[id]?.model.busy = true
        for promise in promises {
            promise.receivePromisedFiles(atDestination: folder, options: [:], operationQueue: .main) { [weak self] url, error in
                MainActor.assumeIsolated {
                    self?.controllers[id]?.model.busy = false
                    if let error {
                        self?.controllers[id]?.model.show(error.localizedDescription, isError: true)
                    } else {
                        self?.importPayloads([.ownedFiles([url])], into: id)
                    }
                }
            }
        }
    }

    // MARK: Acciones

    func run(_ action: any ShelfAction, urls: [URL], in id: Shelf.ID) {
        guard !urls.isEmpty, let controller = controllers[id] else { return }
        controller.model.busy = true
        let root = store.itemsRoot
        Task { @MainActor [weak self] in
            defer { controller.model.busy = false }
            do {
                let folder = try OwnedFolders.make(in: root)
                let result = try await Task.detached(priority: .userInitiated) {
                    try await action.perform(urls, output: folder)
                }.value
                if result.files.isEmpty { try? FileManager.default.removeItem(at: folder) }
                self?.importPayloads([.ownedFiles(result.files)], into: id)
                if let text = result.copiedText {
                    NSPasteboard.general.clearContents()
                    NSPasteboard.general.setString(text, forType: .string)
                    controller.model.show("Texto copiado al portapapeles")
                } else if result.files.count == 1 {
                    controller.model.show("Listo: \(result.files[0].lastPathComponent)")
                } else if result.files.count > 1 {
                    controller.model.show("Listo: \(result.files.count) archivos")
                }
            } catch {
                controller.model.show(error.localizedDescription, isError: true)
            }
        }
    }
}
```

```swift
// FILE: scripts/make-banner.swift
// Banner del README (2560x760) con el estilo de la familia Poxi.
// Uso: make-banner <icono.png> <estanteria.png> <detalle.png> <salida.png>
import SwiftUI
import AppKit

extension Color {
    init(hex: UInt32) {
        self.init(.sRGB, red: Double((hex >> 16) & 0xFF) / 255, green: Double((hex >> 8) & 0xFF) / 255, blue: Double(hex & 0xFF) / 255)
    }
}

let mint = Color(hex: 0x34E0A8)
let edge = Color(hex: 0x2A3446)
let panelColor = Color(hex: 0x111827)
let hardShadow = Color(hex: 0x04070D)

struct Grid: View {
    var body: some View {
        Canvas { context, size in
            var path = Path()
            stride(from: 0, through: size.width, by: 40).forEach { path.addRect(CGRect(x: $0, y: 0, width: 1, height: size.height)) }
            stride(from: 0, through: size.height, by: 40).forEach { path.addRect(CGRect(x: 0, y: $0, width: size.width, height: 1)) }
            context.fill(path, with: .color(.white.opacity(0.035)))
        }
    }
}

struct PixelBox<Content: View>: View {
    var shadow: CGFloat = 8
    var fill = panelColor
    @ViewBuilder var content: Content

    var body: some View {
        content
            .background(fill)
            .overlay(Rectangle().strokeBorder(edge, lineWidth: 3))
            .background(hardShadow.offset(x: shadow, y: shadow))
    }
}

struct Chip: View {
    let text: String
    let color: Color

    var body: some View {
        PixelBox(shadow: 5, fill: Color(hex: 0x0D1320)) {
            HStack(spacing: 14) {
                Rectangle().fill(color).frame(width: 13, height: 13)
                Text(text)
                    .font(.system(size: 19, weight: .black, design: .rounded))
                    .tracking(2.4)
                    .foregroundStyle(Color(hex: 0xEEF2FB))
            }
            .padding(.horizontal, 20)
            .frame(height: 50)
        }
    }
}

struct Banner: View {
    let icon: NSImage
    let shelf: NSImage
    let detail: NSImage

    var body: some View {
        ZStack(alignment: .topLeading) {
            Color(hex: 0x0A111F)
            Grid()
            RadialGradient(colors: [mint.opacity(0.08), .clear], center: .bottomTrailing, startRadius: 0, endRadius: 700)

            PixelBox {
                VStack(alignment: .leading, spacing: 0) {
                    HStack(spacing: 26) {
                        Image(nsImage: icon).resizable().interpolation(.none).frame(width: 150, height: 150).padding(-20)
                        (Text("Poxi").foregroundColor(Color(hex: 0xF4F6FB)) + Text("Drop").foregroundColor(mint))
                            .font(.system(size: 112, weight: .black, design: .rounded))
                            .shadow(color: hardShadow, radius: 0, x: 6, y: 6)
                    }
                    Text("Agita el ratón, suelta y llévalo a donde quieras")
                        .font(.system(size: 30, weight: .bold, design: .rounded))
                        .foregroundStyle(Color(hex: 0x9AA6BD))
                        .padding(.top, 26)
                    VStack(alignment: .leading, spacing: 20) {
                        HStack(spacing: 20) {
                            Chip(text: "AGITAR PARA ABRIR", color: mint)
                            Chip(text: "ACCIONES RÁPIDAS", color: Color(hex: 0x60A5FA))
                        }
                        HStack(spacing: 20) {
                            Chip(text: "GRATIS", color: Color(hex: 0xFFCE3A))
                            Chip(text: "SIN ANUNCIOS", color: Color(hex: 0xF472B6))
                            Chip(text: "OPEN SOURCE", color: Color(hex: 0xA78BFA))
                        }
                    }
                    .padding(.top, 44)
                }
                .padding(.horizontal, 54)
                .padding(.vertical, 50)
                .frame(width: 1110, alignment: .leading)
            }
            .offset(x: 82, y: 82)

            PixelBox(shadow: 10, fill: Color(hex: 0x0A0E1C)) {
                ZStack(alignment: .top) {
                    LinearGradient(colors: [Color(hex: 0x5C4DD9), Color(hex: 0xED739E)], startPoint: .topLeading, endPoint: .bottomTrailing)
                    VStack(spacing: 0) {
                        Image(nsImage: shelf).resizable().aspectRatio(contentMode: .fit).frame(height: 250)
                        Image(nsImage: detail).resizable().aspectRatio(contentMode: .fit).frame(height: 330).offset(y: -40)
                    }
                    .padding(.top, 10)
                }
                .frame(width: 1080, height: 600)
                .clipped()
            }
            .offset(x: 1376, y: 70)

            PixelBox(shadow: 7, fill: mint) {
                Text("AGITA Y SUELTA")
                    .font(.system(size: 34, weight: .black, design: .rounded))
                    .tracking(2)
                    .foregroundStyle(Color(hex: 0x0A111F))
                    .padding(.horizontal, 30)
                    .frame(height: 82)
            }
            .offset(x: 2040, y: 610)
        }
        .frame(width: 2560, height: 760)
        .clipped()
    }
}

@main
enum Main {
    @MainActor static func main() {
        _ = NSApplication.shared
        let args = CommandLine.arguments
        guard args.count == 5, let icon = NSImage(contentsOfFile: args[1]),
              let shelf = NSImage(contentsOfFile: args[2]), let detail = NSImage(contentsOfFile: args[3]) else {
            print("Uso: make-banner <icono.png> <estanteria.png> <detalle.png> <salida.png>")
            exit(1)
        }
        let renderer = ImageRenderer(content: Banner(icon: icon, shelf: shelf, detail: detail))
        renderer.scale = 1
        guard let tiff = renderer.nsImage?.tiffRepresentation,
              let png = NSBitmapImageRep(data: tiff)?.representation(using: .png, properties: [:]) else {
            print("No se pudo renderizar")
            exit(1)
        }
        try! png.write(to: URL(fileURLWithPath: args[4]))
        print("Banner: \(args[4])")
    }
}
```
